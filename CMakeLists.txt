cmake_minimum_required(VERSION 3.24)
project(helium_drm_fixer VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Supported platforms: Windows 11 and Linux")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
    message(FATAL_ERROR "Only x86_64 builds are currently supported")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INSTALL_PROJECT OFF CACHE BOOL "" FORCE)
# HTTPS archives and hashes, deliberately no Git operations.
set(JSON_URL "https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0")
set(MINIZ_URL "https://codeload.github.com/richgel999/miniz/tar.gz/refs/tags/3.1.2")
if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/json-3.12.0.tar.gz")
    set(JSON_URL "${CMAKE_SOURCE_DIR}/.deps/json-3.12.0.tar.gz")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/miniz-3.1.2.tar.gz")
    set(MINIZ_URL "${CMAKE_SOURCE_DIR}/.deps/miniz-3.1.2.tar.gz")
endif()
FetchContent_Declare(json URL "${JSON_URL}"
    URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187)
FetchContent_Declare(miniz URL "${MINIZ_URL}"
    URL_HASH SHA256=98468f8924934b723276680f85238b6c78bf1f8b49b4459cc9b7214a20e2e9fb)
FetchContent_MakeAvailable(json miniz)
target_compile_definitions(miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(fixer_core src/core.cpp src/status.cpp src/operation.cpp src/platform.cpp src/directory.cpp src/network.cpp src/package.cpp src/install.cpp)
target_include_directories(fixer_core PUBLIC include)
target_link_libraries(fixer_core PUBLIC nlohmann_json::nlohmann_json PRIVATE miniz)
target_compile_definitions(fixer_core PUBLIC FIXER_VERSION="${PROJECT_VERSION}")
if(WIN32)
    target_compile_definitions(fixer_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(fixer_core PRIVATE winhttp bcrypt crypt32 advapi32 shell32 ws2_32)
else()
    find_package(CURL 7.85 REQUIRED)
    find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
    target_link_libraries(fixer_core PRIVATE CURL::libcurl OpenSSL::Crypto)
endif()
if(MSVC)
    target_compile_options(fixer_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(fixer_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
option(FIXER_SANITIZERS "Address and undefined behaviour sanitizers (Linux GCC/Clang)" OFF)
if(FIXER_SANITIZERS AND NOT WIN32)
    target_compile_options(fixer_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fixer_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(helium_drm_fixer main.cpp)
target_link_libraries(helium_drm_fixer PRIVATE fixer_core)
include(CTest)
option(FIXER_REQUIRE_NETWORK_TESTS "Require Python/OpenSSL and native HTTPS test target" OFF)
if(BUILD_TESTING)
    add_executable(fixer_tests tests/tests.cpp)
    target_link_libraries(fixer_tests PRIVATE fixer_core miniz)
    add_test(NAME security_and_transactions COMMAND fixer_tests)
    add_test(NAME cli_help COMMAND helium_drm_fixer --help)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli_workflows COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_SOURCE_DIR}/tests/cli_integration.py"
            "$<TARGET_FILE:helium_drm_fixer>" "$<TARGET_FILE:fixer_tests>")
        set_tests_properties(cli_workflows PROPERTIES TIMEOUT 120)
        add_test(NAME installation_detection COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_SOURCE_DIR}/tests/detection_integration.py"
            "$<TARGET_FILE:helium_drm_fixer>" "$<TARGET_FILE:fixer_tests>")
        set_tests_properties(installation_detection PROPERTIES TIMEOUT 120)
        add_test(NAME process_crash_recovery COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/crash_recovery.py" "$<TARGET_FILE:fixer_tests>")
        find_program(FIXER_OPENSSL_EXECUTABLE NAMES openssl
            HINTS "C:/Program Files/Git/mingw64/bin" "C:/Program Files/Git/usr/bin")
        if(FIXER_OPENSSL_EXECUTABLE)
            # Compile the native transport separately: fixture routing/trust must
            # never be enabled in fixer_core or the installed executable.
            add_executable(fixer_network_tests tests/network_probe.cpp src/network.cpp)
            target_compile_definitions(fixer_network_tests PRIVATE FIXER_NETWORK_TESTING)
            target_include_directories(fixer_network_tests PRIVATE tests)
            target_link_libraries(fixer_network_tests PRIVATE fixer_core)
            if(NOT WIN32)
                target_link_libraries(fixer_network_tests PRIVATE CURL::libcurl)
            endif()
            add_test(NAME native_https_failures COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_SOURCE_DIR}/tests/network_integration.py"
                --probe "$<TARGET_FILE:fixer_network_tests>"
                --production "$<TARGET_FILE:helium_drm_fixer>"
                --openssl "${FIXER_OPENSSL_EXECUTABLE}")
            set_tests_properties(native_https_failures PROPERTIES TIMEOUT 120)
        else()
            message(WARNING "OpenSSL executable missing: native HTTPS integration tests unavailable")
        endif()
    endif()
endif()
if(FIXER_REQUIRE_NETWORK_TESTS AND NOT TARGET fixer_network_tests)
    message(FATAL_ERROR "Native HTTPS tests require BUILD_TESTING=ON, Python3 and the OpenSSL executable")
endif()
if(MINGW)
    # CLion's bundled MinGW runtime is shipped alongside the executable.
    get_filename_component(MINGW_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(runtime libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
        foreach(target helium_drm_fixer fixer_tests fixer_network_tests)
            if(TARGET ${target} AND EXISTS "${MINGW_BIN}/${runtime}")
                add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${MINGW_BIN}/${runtime}" "$<TARGET_FILE_DIR:${target}>")
            endif()
        endforeach()
        if(EXISTS "${MINGW_BIN}/${runtime}")
            install(FILES "${MINGW_BIN}/${runtime}" DESTINATION bin)
        endif()
    endforeach()
endif()
install(TARGETS helium_drm_fixer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/helium-drm-fixer)
# Local documentation is intentionally ignored by Git and absent in CI checkouts.
install(FILES SECURITY.md THIRD_PARTY.md DESTINATION share/helium-drm-fixer OPTIONAL)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION share/helium-drm-fixer/licenses RENAME nlohmann-json-MIT.txt)
install(FILES "${miniz_SOURCE_DIR}/LICENSE" DESTINATION share/helium-drm-fixer/licenses RENAME miniz-MIT.txt)
option(FIXER_FUZZING "Build libFuzzer parser harness (Linux Clang only)" OFF)
if(FIXER_FUZZING)
    if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "FIXER_FUZZING requires Linux Clang")
    endif()
    target_compile_options(fixer_core PRIVATE -fsanitize=fuzzer-no-link)
    target_compile_options(miniz PRIVATE -fsanitize=address,undefined,fuzzer-no-link)
    add_executable(fixer_fuzz tests/fuzz_parsers.cpp)
    target_link_libraries(fixer_fuzz PRIVATE fixer_core)
    target_compile_options(fixer_fuzz PRIVATE -fsanitize=fuzzer)
    target_link_options(fixer_fuzz PRIVATE -fsanitize=fuzzer)
endif()
